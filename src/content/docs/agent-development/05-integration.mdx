### 1. Error Handling

LLM çağrıları başarısız olur. API'ler timeout verir. Parser'lar beklenmedik çıktıda boğulur. Agent'ınız bunların hepsini yönetir ya da gece 3'te çöker.

```python
from langgraph.graph import StateGraph, START, END
import logging

logger = logging.getLogger("agent")

class AgentState(TypedDict):
    messages: Annotated[list, add_messages]
    task: str
    result: str
    error: str
    retry_count: int

async def call_external_api(state: AgentState) -> dict:
    """Call an external API with retry logic."""
    max_retries = 3
    retry_count = state.get("retry_count", 0)

    try:
        result = await flight_api.search(
            origin=state["origin"],
            destination=state["destination"]
        )
        return {"result": result, "error": "", "retry_count": 0}

    except TimeoutError:
        if retry_count < max_retries:
            logger.warning(f"API timeout, retry {retry_count + 1}/{max_retries}")
            return {"retry_count": retry_count + 1, "error": "timeout"}
        else:
            logger.error("API timeout after max retries")
            return {"error": "API unavailable after 3 retries", "result": ""}

    except Exception as e:
        logger.error(f"Unexpected error: {e}")
        return {"error": str(e), "result": ""}


def route_after_api_call(state: AgentState) -> str:
    if state.get("error") == "timeout" and state.get("retry_count", 0) < 3:
        return "retry"
    elif state.get("error"):
        return "handle_error"
    return "process_result"
```

<Aside type="tip">
Use **exponential backoff** for retries. Don't hammer a failing service with immediate retries — you make the problem worse.

```python
import asyncio

async def retry_with_backoff(func, max_retries=3, base_delay=1.0):
    for attempt in range(max_retries):
        try:
            return await func()
        except Exception:
            if attempt == max_retries - 1:
                raise
            delay = base_delay * (2 ** attempt)  # 1s, 2s, 4s
            await asyncio.sleep(delay)
```
</Aside>

### 2. Cost Control

Döngüdeki bir agent LLM bütçenizi dakikalar içinde bitirebilir. Kesin limitler şart.

```python
import time
from dataclasses import dataclass, field

@dataclass
class CostGuard:
    """Prevent runaway LLM costs."""
    max_llm_calls: int = 20
    max_cost_usd: float = 5.0
    max_duration_seconds: float = 300

    llm_calls: int = field(default=0, init=False)
    estimated_cost: float = field(default=0.0, init=False)
    start_time: float = field(default_factory=time.time, init=False)

    def check(self, input_tokens: int = 0, output_tokens: int = 0):
        self.llm_calls += 1
        self.estimated_cost += (input_tokens * 0.003 + output_tokens * 0.015) / 1000

        if self.llm_calls > self.max_llm_calls:
            raise CostLimitExceeded(
                f"LLM call limit exceeded: {self.llm_calls}/{self.max_llm_calls}")

        if self.estimated_cost > self.max_cost_usd:
            raise CostLimitExceeded(
                f"Cost limit exceeded: ${self.estimated_cost:.2f}/${self.max_cost_usd}")

        elapsed = time.time() - self.start_time
        if elapsed > self.max_duration_seconds:
            raise CostLimitExceeded(
                f"Duration limit: {elapsed:.0f}s/{self.max_duration_seconds}s")

class CostLimitExceeded(Exception):
    pass
```

```python
# Wire it into your graph nodes
cost_guard = CostGuard(max_llm_calls=15, max_cost_usd=2.0)

async def analyze_with_guard(state: AgentState) -> dict:
    try:
        cost_guard.check()
        response = await model.ainvoke(messages)
        cost_guard.check(
            input_tokens=response.usage_metadata["input_tokens"],
            output_tokens=response.usage_metadata["output_tokens"]
        )
        return {"result": response.content}
    except CostLimitExceeded as e:
        return {"error": f"Safety limit: {e}", "result": ""}
```

<KeyInsight>
Her production agent'ta üç kesin limit: invocation başına max LLM çağrısı, dolar cinsinden max maliyet, max duvar saati süresi. Herhangi bir limit aşılırsa agent zarif biçimde durur. İstisna yok. "Bir çağrı daha" yok.
</KeyInsight>

#### Prompt Injection Defense

En kritik risk. Saldırgan, LLM'in davranışını ele geçiren bir input hazırlar.

```python
async def sanitize_input(state: AgentState) -> dict:
    """Check user input for potential prompt injection."""
    red_flags = [
        "ignore your previous instructions",
        "ignore all previous",
        "you are now",
        "new instructions:",
        "system prompt:",
        "forget everything",
    ]

    input_lower = state["task"].lower()
    for flag in red_flags:
        if flag in input_lower:
            return {
                "error": "Input flagged for potential prompt injection.",
                "task": ""
            }
    return {}
```

<Aside type="caution">
Keyword matching alone is not production-grade. Use a layered approach: input sanitization + LLM-based classification + output validation + tool permission scoping. Defense in depth. No single layer is sufficient.
</Aside>

#### Tool Permission Scoping

Injection olmasa bile LLM tool'ları yanlış kullanabilir. Permission'ları açıkça kapsamlayın.

```python
TOOL_PERMISSIONS = {
    "search_flights": {"allowed": True, "requires_approval": False},
    "create_booking": {"allowed": True, "requires_approval": True},
    "modify_booking": {"allowed": True, "requires_approval": True},
    "delete_booking": {"allowed": False, "requires_approval": True},
    "access_pii":     {"allowed": False, "requires_approval": True},
}

async def tool_gatekeeper(state: AgentState) -> dict:
    """Validate tool calls before execution."""
    tool_name = state["pending_tool_call"]["name"]
    permissions = TOOL_PERMISSIONS.get(tool_name, {"allowed": False})

    if not permissions["allowed"]:
        return {
            "error": f"Tool '{tool_name}' is not permitted.",
            "pending_tool_call": None
        }
    if permissions["requires_approval"]:
        return {"needs_human_approval": True}
    return {}
```

## Capstone: End-to-End Agent

Module 1-4'ten her şey, eksiksiz bir agent'a bağlanmış. Dosya okur, içerik analiz eder, yapılandırılmış raporlar üretir ve hataları zarif biçimde yönetir.

```python
from typing import Annotated
from typing_extensions import TypedDict
from langgraph.graph import StateGraph, START, END
from langgraph.graph.message import add_messages
from langgraph.checkpoint.memory import MemorySaver
from langchain_anthropic import ChatAnthropic
from langchain_core.prompts import ChatPromptTemplate

class AnalysisState(TypedDict):
    messages: Annotated[list, add_messages]
    file_path: str
    file_content: str
    analysis: str
    report: str
    quality_score: int
    error: str

model = ChatAnthropic(model="claude-sonnet-4-20250514", temperature=0)

async def read_file(state: AnalysisState) -> dict:
    """Read the target file into state."""
    try:
        with open(state["file_path"], "r") as f:
            content = f.read()

        if not content.strip():
            return {"error": "File is empty", "file_content": ""}

        max_chars = 50000
        if len(content) > max_chars:
            content = content[:max_chars] + f"\n\n[truncated, {len(content)} total chars]"

        return {"file_content": content, "error": ""}

    except FileNotFoundError:
        return {"error": f"File not found: {state['file_path']}", "file_content": ""}
    except Exception as e:
        return {"error": f"Error reading file: {e}", "file_content": ""}


async def analyze_content(state: AnalysisState) -> dict:
    """Analyze file content with the LLM."""
    prompt = ChatPromptTemplate.from_messages([
        ("system", """You are a senior technical analyst. Analyze the file covering:
1. **Overview**: What is this file? Purpose?
2. **Key Findings**: 3-5 most important observations
3. **Structure**: How is the content organized?
4. **Quality Assessment**: Rate 1-10 with justification
5. **Recommendations**: 2-3 actionable improvements

Reference actual content from the file."""),
        ("human", "Analyze this file ({file_path}):\n\n{file_content}"),
    ])

    chain = prompt | model
    response = await chain.ainvoke({
        "file_path": state["file_path"],
        "file_content": state["file_content"][:30000]
    })
    return {"analysis": response.content}


async def generate_report(state: AnalysisState) -> dict:
    """Generate a structured Markdown report."""
    prompt = ChatPromptTemplate.from_messages([
        ("system", """Transform the analysis into a professional Markdown report:
- Executive Summary (2-3 sentences)
- Key Findings (bullet points)
- Detailed Analysis
- Recommendations (actionable, prioritized)
- Quality Score (1-10)"""),
        ("human", "File: {file_path}\nAnalysis:\n{analysis}"),
    ])

    chain = prompt | model
    response = await chain.ainvoke({
        "file_path": state["file_path"],
        "analysis": state["analysis"]
    })

    quality_score = 7
    for line in response.content.split("\n"):
        if "quality score" in line.lower() and any(c.isdigit() for c in line):
            digits = [int(c) for c in line if c.isdigit()]
            if digits and 1 <= digits[0] <= 10:
                quality_score = digits[0]

    return {
        "report": response.content,
        "quality_score": quality_score,
        "messages": [{"role": "assistant",
                      "content": f"Report generated. Quality: {quality_score}/10"}]
    }


async def handle_error(state: AnalysisState) -> dict:
    return {
        "report": f"## Error\n\nCould not complete analysis: {state['error']}",
        "messages": [{"role": "assistant", "content": f"Error: {state['error']}"}]
    }

def route_after_read(state: AnalysisState) -> str:
    return "handle_error" if state.get("error") else "analyze_content"

# Build the graph
workflow = StateGraph(AnalysisState)
workflow.add_node("read_file", read_file)
workflow.add_node("analyze_content", analyze_content)
workflow.add_node("generate_report", generate_report)
workflow.add_node("handle_error", handle_error)

workflow.add_edge(START, "read_file")
workflow.add_conditional_edges(
    "read_file", route_after_read,
    {"analyze_content": "analyze_content", "handle_error": "handle_error"}
)
workflow.add_edge("analyze_content", "generate_report")
workflow.add_edge("generate_report", END)
workflow.add_edge("handle_error", END)

checkpointer = MemorySaver()
agent = workflow.compile(checkpointer=checkpointer)
```

### Deployment Architecture

A production agent runs as a backend service with persistent state and observability:

```python
from fastapi import FastAPI
from langgraph.checkpoint.postgres.aio import AsyncPostgresSaver

app = FastAPI()

@app.post("/analyze")
async def analyze_file(request: AnalysisRequest):
    async with AsyncPostgresSaver.from_conn_string(DATABASE_URL) as checkpointer:
        agent = workflow.compile(checkpointer=checkpointer)
        config = {"configurable": {"thread_id": request.session_id}}
        result = await agent.ainvoke(
            {"file_path": request.file_path, "messages": []},
            config
        )
        return {"report": result["report"], "quality": result["quality_score"]}

@app.post("/analyze/{session_id}/resume")
async def resume_analysis(session_id: str, approval: bool):
    """Resume a paused graph after human review."""
    async with AsyncPostgresSaver.from_conn_string(DATABASE_URL) as checkpointer:
        agent = workflow.compile(checkpointer=checkpointer)
        config = {"configurable": {"thread_id": session_id}}
        if not approval:
            return {"status": "cancelled"}
        result = await agent.ainvoke(None, config)
        return {"report": result["report"]}
```

### Model Tiering — The Cost Multiplier

Use cheap models for simple tasks, powerful models for complex ones:

| Strategy | Implementation | Cost Impact |
|---|---|---|
| **Model tiering** | Haiku for classification, Sonnet for analysis | 60-80% reduction |
| **Caching** | Cache identical tool calls and LLM responses | 30-50% reduction |
| **Early termination** | Stop graph on error at any node | Prevents waste |
| **Token budgets** | Max tokens per node, per graph, per user | Predictable costs |
